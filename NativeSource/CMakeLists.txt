cmake_minimum_required(VERSION 3.24)
project(StreamVideoPlayerNative VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

option(SVP_BUILD_PLUGIN "Build the Windows D3D11 Unity plugin" ON)
option(SVP_BUILD_VIDEO "Build FFmpeg D3D11 playback (requires a local FFmpeg SDK)" OFF)
if(SVP_BUILD_PLUGIN)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The initial native plugin supports Windows x64 only.")
    endif()
    set(UNITY_PLUGIN_API_DIR "" CACHE PATH "Unity Editor/Data/PluginAPI directory")
    if(NOT EXISTS "${UNITY_PLUGIN_API_DIR}/IUnityGraphicsD3D11.h")
        message(FATAL_ERROR "Set UNITY_PLUGIN_API_DIR to the installed Unity Editor/Data/PluginAPI directory.")
    endif()
    add_library(StreamVideoPlayerNative SHARED src/Plugin.cpp)
    target_include_directories(StreamVideoPlayerNative PRIVATE "${UNITY_PLUGIN_API_DIR}" PUBLIC include)
    target_compile_definitions(StreamVideoPlayerNative PRIVATE SVP_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(StreamVideoPlayerNative PRIVATE d3d11)
    if(MSVC)
        target_compile_options(StreamVideoPlayerNative PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
endif()

if(SVP_BUILD_VIDEO)
    if(NOT SVP_BUILD_PLUGIN)
        message(FATAL_ERROR "Video playback currently requires the Windows plugin build.")
    endif()
    set(FFMPEG_ROOT "" CACHE PATH "FFmpeg shared SDK root")
    if(NOT EXISTS "${FFMPEG_ROOT}/include/libavcodec/avcodec.h")
        message(FATAL_ERROR "Run BuildScripts/Get-FFmpeg.ps1 and set FFMPEG_ROOT.")
    endif()
    add_library(SvpVideoCore STATIC src/VideoSession.cpp)
    target_include_directories(SvpVideoCore PUBLIC include)
    target_include_directories(SvpVideoCore SYSTEM PUBLIC "${FFMPEG_ROOT}/include")
    target_compile_definitions(SvpVideoCore PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(SvpVideoCore PUBLIC d3d11 dxgi "${FFMPEG_ROOT}/lib/avcodec.lib"
        "${FFMPEG_ROOT}/lib/avformat.lib" "${FFMPEG_ROOT}/lib/avutil.lib")
    add_library(StreamVideoPlayerFFmpeg SHARED src/VideoPlugin.cpp)
    target_include_directories(StreamVideoPlayerFFmpeg PRIVATE "${UNITY_PLUGIN_API_DIR}")
    target_compile_definitions(StreamVideoPlayerFFmpeg PRIVATE SVP_EXPORTS)
    target_link_libraries(StreamVideoPlayerFFmpeg PRIVATE SvpVideoCore)
    if(MSVC)
        target_compile_options(SvpVideoCore PRIVATE /W4 /WX /permissive- /utf-8)
        target_compile_options(StreamVideoPlayerFFmpeg PRIVATE /W4 /WX /permissive- /utf-8)
    endif()
    add_executable(VideoSmoke tests/VideoSmoke.cpp)
    target_link_libraries(VideoSmoke PRIVATE SvpVideoCore)
    add_executable(VideoBenchmark tests/VideoBenchmark.cpp)
    target_link_libraries(VideoBenchmark PRIVATE SvpVideoCore psapi winmm)
    file(GLOB FFMPEG_RUNTIME_DLLS "${FFMPEG_ROOT}/bin/*.dll")
    foreach(dll IN LISTS FFMPEG_RUNTIME_DLLS)
        add_custom_command(TARGET VideoSmoke POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${dll}" "$<TARGET_FILE_DIR:VideoSmoke>")
    endforeach()
endif()

if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(LatestFrameSlotTests tests/LatestFrameSlotTests.cpp)
    target_include_directories(LatestFrameSlotTests PRIVATE include)
    target_link_libraries(LatestFrameSlotTests PRIVATE Threads::Threads)
    add_test(NAME LatestFrameSlot COMMAND LatestFrameSlotTests)
    add_executable(H264ParametersTests tests/H264ParametersTests.cpp)
    target_include_directories(H264ParametersTests PRIVATE include)
    add_test(NAME H264Parameters COMMAND H264ParametersTests)
    if(SVP_BUILD_PLUGIN)
        add_executable(NativeApiTests tests/NativeApiTests.cpp)
        target_link_libraries(NativeApiTests PRIVATE StreamVideoPlayerNative)
        add_test(NAME NativeApi COMMAND NativeApiTests)
    endif()
    if(SVP_BUILD_VIDEO)
        add_executable(VideoApiTests tests/VideoApiTests.cpp)
        target_include_directories(VideoApiTests PRIVATE include)
        target_link_libraries(VideoApiTests PRIVATE StreamVideoPlayerFFmpeg)
        add_test(NAME VideoApi COMMAND VideoApiTests)
    endif()
endif()
